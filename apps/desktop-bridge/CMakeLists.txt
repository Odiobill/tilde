set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core DBus Test)
add_library(tilde_notifications STATIC NotificationBridge.cpp NotificationBridge.h)
target_include_directories(tilde_notifications PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(tilde_notifications PUBLIC Qt6::Core Qt6::DBus)
qt_add_executable(tilde-desktop-bridge main.cpp)
target_link_libraries(tilde-desktop-bridge PRIVATE tilde_notifications)
if(BUILD_TESTING)
 qt_add_executable(tilde-notification-fixture tests/service-fixture.cpp)
 target_link_libraries(tilde-notification-fixture PRIVATE Qt6::Core Qt6::DBus)
 qt_add_executable(tst_notifications tests/tst_notifications.cpp)
 target_link_libraries(tst_notifications PRIVATE tilde_notifications Qt6::Test)
 target_compile_definitions(tst_notifications PRIVATE DESKTOP_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol-v1.json" DESKTOP_HELPER_PATH="$<TARGET_FILE:tilde-desktop-bridge>")
 add_test(NAME desktop_notifications COMMAND dbus-run-session -- $<TARGET_FILE:tst_notifications>)
 set_tests_properties(desktop_notifications PROPERTIES TIMEOUT 30)
endif()
