cmake_minimum_required(VERSION 3.24)
project(Tilde VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(TILDE_BUILD_GUI "Build product Qt GUI" OFF)
find_program(TILDE_NODE_EXECUTABLE NAMES node REQUIRED)
execute_process(COMMAND "${TILDE_NODE_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/release/check-runtime.mjs" "${NODE_INCLUDE_DIR}" "${UV_INCLUDE_DIR}"
 WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" RESULT_VARIABLE TILDE_RUNTIME_RESULT OUTPUT_VARIABLE TILDE_RUNTIME_OUTPUT)
if(NOT TILDE_RUNTIME_RESULT EQUAL 0)
 message(FATAL_ERROR "Selected Node/runtime headers are incompatible")
endif()
string(JSON NODE_INCLUDE_DIR GET "${TILDE_RUNTIME_OUTPUT}" headers nodeIncludeDir)
string(JSON UV_INCLUDE_DIR GET "${TILDE_RUNTIME_OUTPUT}" headers uvIncludeDir)
add_subdirectory(packages/transport/native)
if(TILDE_BUILD_GUI)
 add_subdirectory(apps/gui)
 add_subdirectory(apps/desktop-bridge)
endif()
add_subdirectory(packages/tools/native)
file(READ "${CMAKE_SOURCE_DIR}/packaging/release/install-layout.json" TILDE_INSTALL_LAYOUT)
if(TILDE_BUILD_GUI)
 foreach(key gui helper)
  string(JSON destination GET "${TILDE_INSTALL_LAYOUT}" paths ${key})
  get_filename_component(destination_directory "${destination}" DIRECTORY)
  if(key STREQUAL "gui")
   install(TARGETS tilde-gui RUNTIME DESTINATION "${destination_directory}" COMPONENT Runtime)
  else()
   install(TARGETS tilde-desktop-bridge RUNTIME DESTINATION "${destination_directory}" COMPONENT Runtime)
  endif()
 endforeach()
endif()
string(JSON app_root GET "${TILDE_INSTALL_LAYOUT}" paths appRoot)
install(TARGETS tilde_peer LIBRARY DESTINATION "${app_root}/packages/transport/.build" COMPONENT Runtime)
install(TARGETS tilde_scoped_fs LIBRARY DESTINATION "${app_root}/packages/tools/.build" COMPONENT Runtime)
